+@app.post("/upload")
+async def upload_document(file: UploadFile = File(...)):
+
+    temporary_path = None
+
+    try:
+
+        # 1. Validate filename
+
+        filename = (file.filename or "").replace("\\", "/")
+        filename = Path(filename).name
+
+        if not filename:
+            raise HTTPException(
+                status_code=400,
+                detail="Invalid filename."
+            )
+
+        if not filename.lower().endswith(".pdf"):
+            raise HTTPException(
+                status_code=400,
+                detail="Only PDF files are supported."
+            )
+
+        # 2. Read uploaded file
+
+        file_content = await file.read()
+
+        MAX_FILE_SIZE = 25 * 1024 * 1024
+
+        if not file_content:
+            raise HTTPException(
+                status_code=400,
+                detail="Uploaded file is empty."
+            )
+
+        if len(file_content) > MAX_FILE_SIZE:
+            raise HTTPException(
+                status_code=413,
+                detail="PDF size must not exceed 25 MB."
+            )
+
+        # 3. Validate PDF signature
+
+        if not file_content.startswith(b"%PDF-"):
+            raise HTTPException(
+                status_code=400,
+                detail="Invalid PDF file."
+            )
+
+        # 4. Save temporarily for PDF processing
+
+        with tempfile.NamedTemporaryFile(
+            suffix=".pdf",
+            delete=False
+        ) as temporary_file:
+
+            temporary_file.write(file_content)
+
+            temporary_path = temporary_file.name
+
+        # 5. Extract PDF text and metadata
+
+        pdf_result = process_pdf(temporary_path)
+
+        pages = pdf_result.get("pages", [])
+
+        if not pages:
+            raise HTTPException(
+                status_code=422,
+                detail="No readable pages found in this PDF."
+            )
+
+        # 6. Create text chunks
+
+        chunks = create_chunks(
+            pages=pages,
+            chunk_size=1000,
+            overlap=150
+        )
+
+        if not chunks:
+            raise HTTPException(
+                status_code=422,
+                detail="No readable text found. This may be a scanned PDF."
+            )
+
+        # 7. Generate embeddings
+
+        texts = [
+            chunk["text"]
+            for chunk in chunks
+        ]
+
+        embeddings = generate_embeddings(texts)
+
+        # 8. Store chunks and embeddings in ChromaDB
+
+        vector_store = get_vector_store()
+
+        document_id = vector_store.generate_document_id(
+            filename
+        )
+
+        vector_store.add_documents(
+            chunks=chunks,
+            embeddings=embeddings,
+            filename=filename,
+            document_id=document_id
+        )
+
+        # 9. Generate chunk statistics
+
+        statistics = get_chunk_statistics(chunks)
+
+        # 10. Return upload information
+
+        upload_data = {
+            "filename": filename,
+            "document_id": document_id,
+            "file_size_kb": round(
+                len(file_content) / 1024, 2
+            ),
+            "total_pages": pdf_result.get(
+                "total_pages", len(pages)
+            ),
+            "total_chunks": statistics["total_chunks"],
+            "total_characters": statistics["total_characters"],
+            "average_chunk_size": statistics["average_chunk_size"],
+            "embedding_dimensions": int(embeddings.shape[1]),
+            "unique_pages": statistics["unique_pages"],
+            "warnings": pdf_result.get("warnings", [])
+        }
+
+        return {
+            "success": True,
+            "message": "PDF uploaded and indexed successfully.",
+            "data": upload_data,
+
+            # Also expose details at the root for frontend compatibility
+            **upload_data
+        }
+
+    except HTTPException:
+        raise
+
+    except ValueError as error:
+
+        raise HTTPException(
+            status_code=400,
+            detail=str(error)
+        )
+
+    except Exception as error:
+
+        raise HTTPException(
+            status_code=500,
+            detail=f"Document processing failed: {str(error)}"
+        )
+
+    finally:
+
+        # Remove temporary PDF after processing
+
+        if temporary_path and os.path.exists(temporary_path):
+            os.remove(temporary_path)
+
+        await file.close()
+
